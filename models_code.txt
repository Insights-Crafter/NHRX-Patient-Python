===== MedicalRecords.py =====

from database.database import Base
from sqlalchemy import Column , Integer ,  ForeignKey, TIMESTAMP, String, DATE

class MedicalRecords(Base):
    __tablename__ = "MedicalRecords"
    recordID = Column(Integer, primary_key=True)
    patientID = Column(Integer, ForeignKey("Patients.patientID"))
    # These targets are teammate-owned tables; their agreed primary keys are
    # doctors.doctorID and hospitals.hospitalID.
    doctorID = Column(Integer, ForeignKey("doctors.doctorID"))
    hospitalID = Column(Integer, ForeignKey("hospitals.hospitalID"))
    recordDATE = Column(TIMESTAMP)
    recordType = Column(String)
    cheif_complaint = Column(String)
    follow_up_date = Column(DATE)

from sqlalchemy import Integer, Column, ForeignKey, Text
from database.database import Base

class PatientAddress(Base):
    __tablename__ = "PatientAddress"
    AddressID = Column(Integer, primary_key=True)
    patientID = Column(Integer, ForeignKey("Patients.patientID", ondelete="CASCADE", onupdate="CASCADE"))
    # Address fields may contain encrypted values, whose length can exceed
    # the plaintext value's length.
    Area = Column(Text)
    HouseNumber = Column(Text)
    Province = Column(Text)
    City = Column(Text)
    Street = Column(Text)
    Block = Column(Text)

===== RecordTests.py =====


from sqlalchemy import Column, Integer, String, Text, DateTime, ForeignKey, UniqueConstraint
from database.database import Base


class MedicalRecordTest(Base):
    __tablename__ = "medical_record_tests"
    __table_args__ = (
        UniqueConstraint("record_id", "test_id", "performed_at", name="uq_record_test_time"),
    )

    id = Column(Integer, primary_key=True, autoincrement=True)
    record_id = Column(
        Integer,
        ForeignKey("MedicalRecords.recordID", ondelete="CASCADE", onupdate="CASCADE"),
        nullable=False,
        index=True,
    )
    test_id = Column(
        Integer,
        ForeignKey("medical_tests.test_id", ondelete="RESTRICT", onupdate="CASCADE"),
        nullable=False,
        index=True,
    )
    performed_at = Column(DateTime(timezone=True))
    result = Column(Text)
    status = Column(String(20), nullable=False, default="pending")  # pending / completed / cancelled
